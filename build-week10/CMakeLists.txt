cmake_minimum_required(VERSION 3.20)
project(week10 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT week10)

# Алдыңғы апталарда жүктелген кітапханаларды қайта қолданамыз.
# Жоба басқа жерге көшірілсе, GLFW/GLM CMake арқылы жүктеледі.
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
foreach(dep glfw glm)
    string(TOUPPER "${dep}" DEP_UPPER)
    if(NOT DEFINED FETCHCONTENT_SOURCE_DIR_${DEP_UPPER})
        foreach(previous ".." "../../computer_grafig" "../../Архив" "../../Mario")
            set(LOCAL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/${previous}/build/_deps/${dep}-src")
            if(EXISTS "${LOCAL_SOURCE}/CMakeLists.txt")
                set(FETCHCONTENT_SOURCE_DIR_${DEP_UPPER} "${LOCAL_SOURCE}")
                break()
            endif()
        endforeach()
    endif()
endforeach()
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glfw glm)
find_package(OpenGL REQUIRED)

add_executable(week10 main.cpp external/glad/src/gl.c)
target_include_directories(week10 PRIVATE external/glad/include)
target_link_libraries(week10 PRIVATE OpenGL::GL glfw glm::glm)
if(MSVC)
    target_compile_options(week10 PRIVATE /W4 /utf-8)
else()
    target_compile_options(week10 PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(camera_tests tests/camera_tests.cpp external/glad/src/gl.c)
    target_include_directories(camera_tests PRIVATE external/glad/include)
    target_link_libraries(camera_tests PRIVATE OpenGL::GL glfw glm::glm)
    if(MSVC)
        target_compile_options(camera_tests PRIVATE /utf-8)
    endif()
    add_test(NAME camera_movement COMMAND camera_tests)
endif()
